  <div id="map"></div>

  <!-- Leaflet JS -->
  <script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>

  <script>
    // Inicializar mapa centrado en Lanús
    var map = L.map('map').setView([-34.707, -58.397], 13);

    // Mapa base OpenStreetMap
    var osm = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      maxZoom: 19,
      attribution: '© OpenStreetMap'
    }).addTo(map);

    // Control de capas
    var overlayMaps = {};

    // 1. Cargar Límite de Lanús
    fetch('Dpto_Lanus.geojson')
      .then(res => res.json())
      .then(data => {
        var layer = L.geoJSON(data, {
          style: { color: '#333', weight: 2, fillOpacity: 0 }
        }).addTo(map);
        overlayMaps["Límite Lanús"] = layer;
        controlCapas.addOverlay(layer, "Límite Lanús");
      });

    // 2. Cobertura Actual EV (10 min)
    fetch('Cobertura_EV.geojson')
      .then(res => res.json())
      .then(data => {
        var layer = L.geoJSON(data, {
          style: { color: '#27ae60', weight: 1, fillColor: '#2ecc71', fillOpacity: 0.4 }
        }).addTo(map);
        controlCapas.addOverlay(layer, "Cobertura actual (10 min)");
      });

    // 3. Cobertura Nueva Propuesta (Sitios Potenciales)
    fetch('Cobertura_SP.geojson')
      .then(res => res.json())
      .then(data => {
        var layer = L.geoJSON(data, {
          style: { color: '#8e44ad', weight: 1, fillColor: '#9b59b6', fillOpacity: 0.4 }
        });
        controlCapas.addOverlay(layer, "Nueva cobertura propuesta");
      });

    // 4. Espacios Verdes Actuales (Plazas y Parques)
    fetch('Espacios_verdes.geojson')
      .then(res => res.json())
      .then(data => {
        var layer = L.geoJSON(data, {
          style: { color: '#1e8449', weight: 1, fillColor: '#1e8449', fillOpacity: 0.8 },
          onEachFeature: function (feature, layer) {
            if (feature.properties && feature.properties.name) {
              layer.bindPopup("<b>Espacio Verde:</b> " + feature.properties.name);
            }
          }
        }).addTo(map);
        controlCapas.addOverlay(layer, "Plazas y Parques Actuales");
      });

    // 5. Sitios Potenciales (Puntos Candidatos)
    fetch('Sitios_potenciales.geojson')
      .then(res => res.json())
      .then(data => {
        var layer = L.geoJSON(data, {
          pointToLayer: function (feature, latlng) {
            return L.circleMarker(latlng, {
              radius: 8,
              fillColor: "#e74c3c",
              color: "#ffffff",
              weight: 2,
              opacity: 1,
              fillOpacity: 0.9
            });
          },
          onEachFeature: function (feature, layer) {
            layer.bindPopup("<b>Sitio Candidato Propuesto</b>");
          }
        }).addTo(map);
        controlCapas.addOverlay(layer, "Sitios Candidatos Propuestos");
      });

    // Agregar control de capas al mapa
    var controlCapas = L.control.layers({ "OpenStreetMap": osm }, null, { collapsed: false }).addTo(map);
  </script>
